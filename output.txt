SELECT
    NS.NTFN_STAT_CODE,
    COUNT(*) AS COUNT
FROM MCS_NTFN N
JOIN MCS_NTFN_STAT NS ON NS.NTFN_ID = N.NTFN_ID
    AND NS.NTFN_HIST_STAT_IND = 'C'
WHERE (
    (NS.NTFN_STAT_CODE = 'C' AND NS.CRTN_TMST >= CURRENT_TIMESTAMP - INTERVAL '1' DAY)
    OR (NS.NTFN_STAT_CODE <> 'C')
)
AND N.REPR_FAC_AREA_ID = ?
